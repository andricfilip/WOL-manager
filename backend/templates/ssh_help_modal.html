Linux (Ubuntu/Debian)

1. {{ t('ssh_help_install_server') }}

sudo apt update
sudo apt install openssh-server -y
sudo systemctl enable ssh
sudo systemctl start ssh

2. {{ t('ssh_help_check_status') }}

sudo systemctl status ssh

{{ t('ssh_help_should_see_running') }}

3. {{ t('ssh_help_create_user') }}

sudo adduser wol-manager
sudo usermod -aG sudo wol-manager

4. {{ t('ssh_help_nopasswd_shutdown') }}

sudo visudo

{{ t('ssh_help_add_to_end') }}

wol-manager ALL=(ALL) NOPASSWD: /sbin/shutdown, /sbin/poweroff

5. {{ t('ssh_help_testing') }}

ssh wol-manager@localhost
sudo shutdown -h now

Windows 10/11

1. {{ t('ssh_help_install_server') }}

{{ t('ssh_help_open_ps_admin') }}:

Add-WindowsCapability -Online -Name OpenSSH.Server~~~~0.0.1.0

2. {{ t('ssh_help_start_service') }}

Start-Service sshd
Set-Service -Name sshd -StartupType 'Automatic'

3. {{ t('ssh_help_firewall_rule') }}

New-NetFirewallRule -Name sshd -DisplayName 'OpenSSH Server (sshd)' -Enabled True -Direction Inbound -Protocol TCP -Action Allow -LocalPort 22

4. {{ t('ssh_help_ps_script') }}

{{ t('ssh_help_create_file') }} C:\Scripts\shutdown.ps1:

Stop-Computer -Force

5. {{ t('ssh_help_testing') }}

ssh username@localhost
powershell -File C:\Scripts\shutdown.ps1
